-- job 14: fwios-portfolio-crypto-daily-snapshot
select net.http_post(
      url := (select decrypted_secret from vault.decrypted_secrets where name='fwios_project_url' limit 1) || '/functions/v1/portfolio-mtm-refresh-v1',
      headers := jsonb_build_object(
        'Content-Type','application/json',
        'x-fwios-automation-token',(select decrypted_secret from vault.decrypted_secrets where name='fwios_automation_token' limit 1)
      ),
      body := jsonb_build_object(
        'source','CRON_US_CLOSE_CRYPTO_SNAPSHOT',
        'stock_mode','OFF',
        'refresh_crypto',true,
        'requested_at',now()
      ),
      timeout_milliseconds := 25000
    )
    where to_char(now() at time zone 'America/New_York','HH24:MI')='17:10';

-- job 15: fwios-portfolio-stock-post-close-sync
select net.http_post(
      url := (select decrypted_secret from vault.decrypted_secrets where name='fwios_project_url' limit 1) || '/functions/v1/portfolio-mtm-refresh-v1',
      headers := jsonb_build_object(
        'Content-Type','application/json',
        'x-fwios-automation-token',(select decrypted_secret from vault.decrypted_secrets where name='fwios_automation_token' limit 1)
      ),
      body := jsonb_build_object(
        'source','CRON_US_CLOSE_STOCK_' || to_char((now() at time zone 'America/New_York')::date,'YYYYMMDD'),
        'stock_mode','DAILY_CLOSE_ONLY',
        'refresh_crypto',false,
        'requested_at',now()
      ),
      timeout_milliseconds := 25000
    )
    where to_char(now() at time zone 'America/New_York','HH24:MI')='17:10'
      and extract(isodow from now() at time zone 'America/New_York') between 1 and 5
      and not exists (
        select 1
        from (
          select distinct asset_symbol
          from fwios.v_portfolio_positions_current
          where upper(source_position_status)='OPEN'
            and coalesce(quantity,0)>0
            and upper(asset_class)='STOCK'
        ) s
        where not exists (
          select 1
          from fwios.market_observations o
          join fwios.v_instrument_identity_current id on id.instrument_id=o.instrument_id
          where id.asset_symbol=s.asset_symbol
            and o.observation_type='REGULAR_CLOSE'
            and o.value_currency='USD'
            and o.provenance_status='PASS'
            and o.source_tier='A'
            and o.session_date=(now() at time zone 'America/New_York')::date
        )
      );
